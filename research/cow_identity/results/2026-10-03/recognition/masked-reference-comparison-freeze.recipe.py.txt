    raise FileExistsError('Never replace a frozen comparison')
encoding_path = root / 'recognition_masked_encoding_protocol.json'
encoding = json.loads(encoding_path.read_text())
files = dict(encoding['files'])
paths = [
    encoding_path,
    Path(__file__),
    root / 'recognition_masked_replay.py',
    root / 'test_recognition_masked_replay.py',
    root / 'cutie_appearance.py',
    root / 'video_assessment.py',
    root / 'scoring.py',

    Path('datasets/8-calves/video/pmfeed_4_3_16.pkl'),
]

    'original60': '.cache/cow-farm-evaluation/step-000/gallery',
    'reviewed39': '.cache/cow-masked-reviewed-gallery-v1',
}

    paths.extend(directory / name for name in ('manifest.json', 'vectors.npz'))

for path, expected in files.items():
    if digest(Path(path)) != expected:
        raise ValueError(f'Previously frozen input changed: {path}')

    'frozen_at_utc': datetime.now(UTC).isoformat(),

    'files': files,

    'encoder_fingerprint': encoding['encoder_fingerprint'],
    'queries': str(queries),
    'galleries': galleries,
    'annotations': 'datasets/8-calves/video/pmfeed_4_3_16.safe-v1.npz',
    'source_pickle': 'datasets/8-calves/video/pmfeed_4_3_16.pkl',

    'success_reference': {'conservative_precision': 0.99, 'all_visible_known_coverage': 0.6, 'unknown_false_naming_rate': 0.01},
    'followup': 'Report both packages regardless of outcome. No retuning/backfill after scoring. This diagnostic alone cannot promote app behavior or reopen3000+video.',
}

print(json.dumps({'path': str(output), 'sha256': digest(output), 'bindings': len(files)}))
