import hashlib, json, os, subprocess, time
from pathlib import Path
import psutil

root = Path.cwd()
log = root / ".cache/cow-autocattlogger/pose-pilot/inference.log"
output = (
    root
    / "research/cow_identity/results/2026-10-03/recognition/autocattlogger-pose-execution.json"
)
env = os.environ.copy()
env.pop("PYTORCH_ENABLE_MPS_FALLBACK", None)
env["PYTHONPATH"] = "/tmp/cow-pose-reference-deps:research/cow_identity"
env["OMP_NUM_THREADS"] = "2"
command = [
    str(root / "detector/.venv/bin/python"),
    "research/cow_identity/autocattlogger_pose_pilot.py",
    "infer",
]
start = time.monotonic()
peak = 0
stopped = None
with log.open("w") as stream:
    child = subprocess.Popen(command, env=env, stdout=stream, stderr=subprocess.STDOUT)
    while child.poll() is None:
        try:
            proc = psutil.Process(child.pid)
            rss = proc.memory_info().rss + sum(
                p.memory_info().rss for p in proc.children(recursive=True)
            )
            peak = max(peak, rss)
        except psutil.NoSuchProcess:
            break
        if time.monotonic() - start > 180 or rss > 8 * 1024**3:
            stopped = "180s wall" if time.monotonic() - start > 180 else "8GiB RSS"
            child.terminate()
            try:
                child.wait(timeout=5)
            except subprocess.TimeoutExpired:
                child.kill()
            break
        time.sleep(0.1)
    code = child.wait()
r = {
    "command": command,
    "env_overrides": {
        "PYTHONPATH": env["PYTHONPATH"],
        "OMP_NUM_THREADS": "2",
        "PYTORCH_ENABLE_MPS_FALLBACK": "unset",
    },
    "exit_code": code,
    "stop_reason": stopped,
    "elapsed_seconds": time.monotonic() - start,
    "rss_sampled_max_bytes": peak,
    "watchdog_source_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "protocol_sha256": hashlib.sha256(
        (
            root / "research/cow_identity/autocattlogger_pose_pilot_protocol.json"
        ).read_bytes()
    ).hexdigest(),
    "scope": "External wall/RSS watchdog; internal per-batch MPS driver check retained; no semantic changes.",
}
output.write_text(json.dumps(r, indent=2) + "\n")
print(json.dumps(r))
raise SystemExit(code)
